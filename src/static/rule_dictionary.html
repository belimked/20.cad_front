规则描述

该规则定义了人员查询时的格式要求，包括查询参数的格式、查询结果的展示格式等。

规则代码
function processPersonQuery(query) {
  // 验证查询参数
  if (!query.personName && !query.personId) {
    return { error: "人员查询需要提供姓名或工号" };
  }
  
  // 格式化查询结果
  return {
    operation: "查询",
    object: "人员",
    personName: query.personName || "无",
    personId: query.personId || "无",
    role: query.role || "无",
    project: query.project || "无"
  };
}
